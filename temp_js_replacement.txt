  /* ── Read Question button ───────────────────────────────────── */
  window.audioReadQuestion = function () {
    var btn = document.getElementById('audio-read-btn');
    var text = document.getElementById('audio-question-text');
    var audioPlayer = document.getElementById('question-audio-player');
    
    if (!text) return;

    // Stop any currently playing audio
    if (audioPlayer && !audioPlayer.paused) {
      audioPlayer.pause();
      audioPlayer.currentTime = 0;
      btn.innerHTML = '<span>&#128266;</span> Read Question';
      return;
    }

    // Stop any Web Speech API synthesis
    if (synth && synth.speaking) {
      synth.cancel();
      btn.innerHTML = '<span>&#128266;</span> Read Question';
      return;
    }

    btn.innerHTML = '<span>&#128266;</span> Reading…';

    // Try to use generated audio first, fallback to Web Speech API
    if (audioPlayer && audioPlayer.src) {
      audioPlayer.onended = function() {
        btn.innerHTML = '<span>&#128266;</span> Read Question';
        audioPlayer.onended = null;
      };
      audioPlayer.onerror = function() {
        // Fallback to Web Speech API if audio fails
        console.warn('Audio playback failed, falling back to Web Speech API');
        speak(text.innerText, function () {
          btn.innerHTML = '<span>&#128266;</span> Read Question';
        });
        audioPlayer.onerror = null;
      };
      audioPlayer.play();
    } else {
      // Fallback to Web Speech API
      speak(text.innerText, function () {
        btn.innerHTML = '<span>&#128266;</span> Read Question';
      });
    }
  };
